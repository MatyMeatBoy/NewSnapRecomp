# (newsnap/overlays/snap_ui_pace.c). Snap 2.0.1 main build ddb900ff04a912ed42430508389853cfaf1accce.
# The interface manager advances every movie one timeline frame per game frame:
#   7838e4 ldr x8,[x22,#0xe0]   start of the first walk of the movie list
#   7838f8 ldr x0,[x20,#0x10]   movie for the call at 7838fc (bl 9d0de0)
#   790de8 ldr x8,[x8,#0xe0]    start of the second walk
#   790dfc ldr x0,[x21,#0x10]   movie for the call at 790e00 (bl 9d0de0)
# The begin helper runs before the original load and changes nothing in the guest. The movie
# helper runs after the original load and, on a frame that is not due, points x0 at an empty
# stand-in the player's first check rejects. AC_SNAP_UI_PACE_PATCH=0 removes all four sites.
set snap_ui_pace AC_SNAP_UI_PACE_PATCH 1
site snap_ui_pace main 7838e4 f94072c8 @fn:snap_ui_pace_begin f94072c8
site snap_ui_pace main 7838f8 f9400a80 f9400a80 @fn:snap_ui_pace_movie
site snap_ui_pace main 790de8 f9407108 @fn:snap_ui_pace_begin f9407108
site snap_ui_pace main 790dfc f9400aa0 f9400aa0 @fn:snap_ui_pace_movie
